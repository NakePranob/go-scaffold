func appError(err error) error {
    switch {
    case errors.Is(err, domain.ErrNotFound):
        return applicationError(http.StatusNotFound, "ORDER_NOT_FOUND", "order not found")
    case errors.Is(err, domain.ErrConflict):
        return applicationError(http.StatusConflict, "ORDER_CONFLICT", "order already exists")
    // ...อีก 3 cases ที่ต้องดูแลแยกทุก module
    default:
        return apperror.NewInternal(err)
    }
}

03 Auth / RBAC

สองโมดูลนี้เคยมี toHTTPError switch คนละชุด — หลังแก้ใช้ appError + ตาราง code→status + apperror.Translate เหมือน module อื่น

ก่อนแก้ · auth handler.go
ปัจจุบัน

หนึ่งในสอง switch ที่ทำให้ patcher ต้องเดาชื่อ mapper

internal/app/user/adapters/inbound/http/handler.go
func toHTTPError(err error) error {
    if err == nil { return nil }
    var appErr *apperror.AppError
    if errors.As(err, &appErr) { return err }

    var ruleErr *domain.RuleError
    if errors.As(err, &ruleErr) {
        status := http.StatusInternalServerError
        switch ruleErr.Code {
        case "USER_NOT_FOUND":
            status = http.StatusNotFound
        case "USER_EMAIL_TAKEN", "AUTH_ALREADY_VERIFIED", "AUTH_MFA_ALREADY_ENABLED":
            status = http.StatusConflict
        // ...อีกหลาย code ใน switch เดียวกัน
        }
        return apperror.New(status, ruleErr.Code, ruleErr.Message)
    }
    switch {
    case errors.Is(err, domain.ErrNotFound):
        return apperror.New(http.StatusNotFound, "USER_NOT_FOUND", "user not found")
    // ...sentinel cases อีกชุด
    }
}

Expected

ชื่อฟังก์ชันเหมือน module อื่น; code→status เป็นตาราง, sentinel ใช้ translator กลาง

internal/app/user/adapters/inbound/http/handler.go
var httpErrorStatus = map[string]int{
    "USER_NOT_FOUND":          http.StatusNotFound,
    "USER_EMAIL_TAKEN":        http.StatusConflict,
    "AUTH_MFA_ALREADY_ENABLED": http.StatusConflict,
    // ...policy table
}

func appError(err error) error {
    if err == nil { return nil }
    var ruleErr *domain.RuleError
    if errors.As(err, &ruleErr) {
        status := httpErrorStatus[ruleErr.Code]
        if status == 0 { status = http.StatusInternalServerError }
        return apperror.New(status, ruleErr.Code, ruleErr.Message)
    }
    return apperror.Translate(err, httpErrors...)
}
!
ข้อจำกัดที่ตั้งใจ: auth/RBAC ยังมี code→status table เพราะ RuleError เป็น code-based ไม่ใช่ sentinel — แต่เหลือ “หนึ่งชื่อ + หนึ่งกลไก” ไม่ใช่ switch ซ้ำอีกชุด

internal/shared/apperror/apperror.go
type Rule struct {
    Err     error
    Status  int
    Code    string
    Message string
}

func Translate(err error, rules ...Rule) error {
    if err == nil { return nil }
    var appErr *AppError
    if errors.As(err, &appErr) { return err }
    for _, rule := range rules {
        if errors.Is(err, rule.Err) {
            return New(rule.Status, rule.Code, rule.Message)
        }
    }
    return NewInternal(err)
}

08 Scope

สิ่งที่ทำในรอบนี้ และสิ่งที่ตั้งใจไม่ทำ

อยู่ใน scope

  • Module handlers: ใช้ httpErrors table + apperror.Translate + httpx.*
  • Auth / RBAC: เลิก toHTTPError switch, ใช้ชื่อ appError เดียวกัน
  • generate method: ใช้ helper เดียวกัน, เลิก regex หาชื่อ mapper
  • Docs: docs/architect/http-contract.md ฉบับเดียว
  • QA: regression + target tiers, external oracle

นอก scope (แยก change)

  • Uniform {data, meta} envelope: breaking กับ auth/MFA และ FSD clients
  • CLI --json: เป็นคนละ layer กับ HTTP API
  • เปลี่ยน auth response fields: access token, MFA challenge, refresh cookie
  • Auto-migrate โปรเจกต์เก่า: template มีผลกับ generation ใหม่เท่านั้น
